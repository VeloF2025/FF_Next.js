```typescript
// scripts/seed-civils-gallery.ts
// Run with: npx tsx scripts/seed-civils-gallery.ts
// Seeds vlm_visual_photo_examples with civil step examples from construction_qa_photos.
// Source of truth: PASS reviews → positive (vlm_valid=true), RETAKE reviews → negative (vlm_valid=false).
// Max 6 positive + 6 negative per step. ON CONFLICT DO NOTHING (idempotent).

import { Pool } from 'pg';

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

const STEP_LABELS: Record<number, string[]> = {
  1: ['Before Photo', '1 - Before Photo'],
  2: ['During Photo'],
  3: ['Depth Photo'],
  4: ['End Plates', '4 - End Plates'],
  5: ['Compaction', 'Compaction / Backfill', '5 - Compaction / Backfill'],
  6: ['Level Check'],
  7: ['After Photo', '7 - After Photo', 'AFTER photo'],
  8: ['Signature'],
};

async function seedStep(stepNumber: number, labels: string[]): Promise<void> {
  const placeholders = labels.map((_, i) => `$${i + 1}`).join(', ');

  // Positive: PASS reviews, vlm_valid=true, canonical label
  const positiveResult = await pool.query<{ storage_url: string; vlm_confidence: number }>(
    `SELECT p.storage_url, p.vlm_confidence
     FROM construction_qa_photos p
     JOIN construction_qa_reviews r ON r.id = p.review_id
     WHERE r.discipline = 'civil'
       AND r.qa_decision = 'PASS'
       AND p.vlm_valid = true
       AND p.checklist_step = ${labels.length + 1}
       AND p.step_label = ANY(ARRAY[${placeholders}])
       AND p.storage_url IS NOT NULL
       AND p.storage_url != ''
     ORDER BY p.vlm_confidence DESC NULLS LAST
     LIMIT 6`,
    labels
  );

  // Negative: RETAKE reviews, vlm_valid=false, canonical label
  const negativeResult = await pool.query<{ storage_url: string; vlm_confidence: number }>(
    `SELECT p.storage_url, p.vlm_confidence
     FROM construction_qa_photos p
     JOIN construction_qa_reviews r ON r.id = p.review_id
     WHERE r.discipline = 'civil'
       AND r.qa_decision = 'RETAKE'
       AND p.vlm_valid = false
       AND p.checklist_step = ${labels.length + 1}
       AND p.step_label = ANY(ARRAY[${placeholders}])
       AND p.storage_url IS NOT NULL
       AND p.storage_url != ''
     ORDER BY p.vlm_confidence DESC NULLS LAST
     LIMIT 6`,
    labels
  );

  let inserted = 0;
  for (const row of positiveResult.rows) {
    const r = await pool.query(
      `INSERT INTO vlm_visual_photo_examples (job_type, step_number, photo_url, label, confidence)
       VALUES ('civils', $1, $2, 'positive', $3)
       ON CONFLICT (photo_url) DO NOTHING`,
      [stepNumber, row.storage_url, row.vlm_confidence ?? null]
    );
    inserted += r.rowCount ?? 0;
  }
  for (const row of negativeResult.rows) {
    const r = await pool.query(
      `INSERT INTO vlm_visual_photo_examples (job_type, step_number, photo_url, label, confidence)
       VALUES ('civils', $1, $2, 'negative', $3)
       ON CONFLICT (photo_url) DO NOTHING`,
      [stepNumber, row.storage_url, row.vlm_confidence ?? null]
    );
    inserted += r.rowCount ?? 0;
  }

  console.log(
    `Step ${stepNumber}: +${positiveResult.rows.length} positive, ` +
    `+${negativeResult.rows.length} negative → ${inserted} inserted`
  );
}

async function main(): Promise<void> {
  console.log('Seeding civils gallery examples…');
  for (const [stepStr, labels] of Object.entries(STEP_LABELS)) {
    await seedStep(Number(stepStr), labels);
  }
  await pool.end();
  console.log('Done.');
}

main().catch((err) => { console.error(err); process.exit(1); });
```

// src/lib/vlmGallery.ts
// Shared utility: load gallery examples from vlm_visual_photo_examples.
// Used by stepQualityValidationService (activation auto-QA) and
// sitecam validate endpoint (both activation + civils).

import { pool } from '@/lib/db';
import { log } from '@/lib/logger';
import { fetchPhotoAsBase64 } from '@/modules/activate/services/photoFetchService';
import type { GalleryExamples } from '@/modules/activate/services/stepQualityCriteria';

const MODULE = 'vlmGallery';

export async function loadGalleryExamples(
  step: number,
  jobType: 'activation' | 'civils'
): Promise<GalleryExamples | undefined> {
  try {
    const { rows } = await pool.query<{ photo_url: string; label: string }>(
      `SELECT photo_url, label
       FROM vlm_visual_photo_examples
       WHERE step_number = $1
         AND job_type = $2
       ORDER BY saved_at DESC
       LIMIT 6`,
      [step, jobType]
    );

    const positiveRows = rows.filter((r) => r.label === 'positive');
    const negativeRows = rows.filter((r) => r.label === 'negative');
    if (positiveRows.length === 0 && negativeRows.length === 0) return undefined;

    const toBase64 = async (url: string): Promise<string | null> => {
      try {
        return await fetchPhotoAsBase64(url);
      } catch (err) {
        log.warn('Failed to fetch gallery example as base64', { url, err: String(err) }, MODULE);
        return null;
      }
    };

    const [positiveResults, negativeResults] = await Promise.all([
      Promise.all(positiveRows.map((r) => toBase64(r.photo_url))),
      Promise.all(negativeRows.map((r) => toBase64(r.photo_url))),
    ]);

    return {
      positiveBase64: positiveResults.filter((b): b is string => b !== null),
      negativeBase64: negativeResults.filter((b): b is string => b !== null),
    };
  } catch (err) {
    log.warn('Failed to load gallery visual examples', { err }, MODULE);
    return undefined;
  }
}

interface ValidateBody {
  jobType: 'activations' | 'civils';
  stepNumber: number;
  siteId: string;
  photoBase64: string;
  attemptNumber: number;
  exifTimestamp?: string;
}

interface VlmResult {
  pass: boolean;
  reasons: string[];
  corrections: string[];
}

function hashBase64(b64: string): string {
  return createHash('sha256').update(b64).digest('hex');
}

async function isDuplicatePhoto(siteId: string, hash: string): Promise<boolean> {
  const { rows } = await pool.query<{ id: string }>(
    `SELECT id FROM pwa_photo_hashes WHERE site_id = $1 AND photo_hash = $2 LIMIT 1`,
    [siteId, hash]
  );
  return rows.length > 0;
}

async function recordPhotoHash(siteId: string, stepNumber: number, hash: string): Promise<void> {
  await pool.query(
    `INSERT INTO pwa_photo_hashes (site_id, step_number, photo_hash)
     VALUES ($1, $2, $3)
     ON CONFLICT DO NOTHING`,
    [siteId, stepNumber, hash]
  );
}

function checkExifAge(exifTimestamp: string | undefined): { ok: boolean; reason?: string } {
  if (!exifTimestamp) return { ok: true };
  const exif = new Date(exifTimestamp).getTime();
  if (isNaN(exif)) return { ok: true };
  const twoHours = 2 * 60 * 60 * 1000;
  if (Math.abs(Date.now() - exif) > twoHours) {
    return { ok: false, reason: 'Photo was taken more than 2 hours ago. Please retake with a fresh photo.' };
  }
  return { ok: true };
}

async function runVlmCheck(
  jobType: 'activations' | 'civils',
  step: number,
  photoBase64: string
): Promise<VlmResult> {
  // Load gallery examples for this job type + step
  const galleryJobType = jobType === 'civils' ? 'civils' : 'activation';
  const galleryExamples = await loadGalleryExamples(step, galleryJobType);

  // Build VLM prompt content
  let content: unknown[];
  if (jobType === 'civils') {
    const { content: c } = buildCivilMessageContent(step as CivilStep, photoBase64, galleryExamples);
    content = c;
  } else {
    const { content: c } = buildMessageContent(step as QualityCheckStep, photoBase64, galleryExamples);
    content = c;
  }

  const body = {
    model: VLM_CATEGORIZATION_MODEL,
    messages: [{ role: 'user', content }],
    max_tokens: VLM_MAX_TOKENS_QUICK,
    temperature: VLM_TEMPERATURE,
  };

  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), VLM_TIMEOUT_REALTIME);
  try {
    const resp = await fetch(VLM_CHAT_ENDPOINT, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
      signal: controller.signal,
    });
    clearTimeout(timeout);
    if (!resp.ok) throw new Error(`VLM HTTP ${resp.status}`);
    const json = await resp.json() as { choices?: Array<{ message?: { content?: string } }> };
    const raw = stripThinkTags(json.choices?.[0]?.message?.content ?? '');
    const match = raw.match(/\{[\s\S]*\}/);
    if (match) {
      const parsed = JSON.parse(match[0]) as { pass?: boolean; passes?: boolean; reasons?: string[]; corrections?: string[]; fail_reason?: string };
      const didPass = parsed.pass === true || parsed.passes === true;
      const failReason = parsed.fail_reason ?? null;
      return {
        pass: didPass,
        reasons: failReason ? [failReason] : (parsed.reasons ?? []),
        corrections: parsed.corrections ?? [],
      };
    }
    return { pass: true, reasons: [], corrections: [] };
  } catch (err) {
    clearTimeout(timeout);
    log.warn('VLM validation failed — allowing pass to avoid blocking technician', { error: String(err) }, MODULE);
    return { pass: true, reasons: [], corrections: [] };
  }
}

async function handler(req: NextApiRequest, res: NextApiResponse): Promise<void> {
  if (req.method !== 'POST') return apiResponse.methodNotAllowed(res, req.method ?? 'UNKNOWN', ['POST']);

  const { jobType, stepNumber, siteId, photoBase64, attemptNumber, exifTimestamp } =
    (req.body ?? {}) as Partial<ValidateBody>;

  if (!jobType || stepNumber == null || !siteId || !photoBase64 || attemptNumber == null) {
    return apiResponse.badRequest(res, 'jobType, stepNumber, siteId, photoBase64, attemptNumber required');
  }

  // Check step has VLM criteria for its job type
  if (jobType === 'civils') {
    if (!(CIVIL_QUALITY_STEPS as readonly number[]).includes(stepNumber)) {
      return apiResponse.badRequest(res, `Civil step ${stepNumber} has no VLM validation criteria`);
    }
  } else {
    if (!(QUALITY_CHECK_STEPS as readonly number[]).includes(stepNumber)) {
      return apiResponse.badRequest(res, `Step ${stepNumber} has no VLM validation criteria`);
    }
  }

  // Get step label for response
  const stepLabel = jobType === 'civils'
    ? CIVIL_STEP_CRITERIA[stepNumber as CivilStep].label
    : STEP_CRITERIA[stepNumber as QualityCheckStep].label;

  // Fraud: EXIF age check
  const ageCheck = checkExifAge(exifTimestamp);
  if (!ageCheck.ok) {
    return apiResponse.success(res, {
      pass: false,
      reasons: [ageCheck.reason!],
      corrections: ['Please take a fresh photo right now at the installation site.'],
      stepLabel,
      attemptNumber,
      maxAttempts: MAX_ATTEMPTS,
      fraudDetected: 'exif_age',
    });
  }

  // Fraud: duplicate photo check
  const hash = hashBase64(photoBase64);
  const duplicate = await isDuplicatePhoto(siteId, hash);
  if (duplicate) {
    return apiResponse.success(res, {
      pass: false,
      reasons: ['This exact photo has already been submitted for this job.'],
      corrections: ['Take a new photo — do not reuse a previously submitted photo.'],
      stepLabel,
      attemptNumber,
      maxAttempts: MAX_ATTEMPTS,
      fraudDetected: 'duplicate_hash',
    });
  }

  const result = await runVlmCheck(jobType, stepNumber, photoBase64);
  await recordPhotoHash(siteId, stepNumber, hash).catch(() => undefined);

  return apiResponse.success(res, {
    pass: result.pass,
    reasons: result.reasons,
    corrections: result.corrections,
    stepLabel,
    attemptNumber,
    maxAttempts: MAX_ATTEMPTS,
  });
}

export default withAuth(handler);

  api: { bodyParser: { sizeLimit: '15mb' } },
};

 *
 * Updates the label (positive/negative) for a civil gallery photo in
 * vlm_visual_photo_examples. Used by the CivilsGalleryTab to toggle good/bad.
 *
 * Auth: requires `manager` role or higher.

import type { NextApiRequest, NextApiResponse } from 'next';
import { withAuth, withRole } from '@/lib/auth';
import { apiResponse } from '@/lib/apiResponse';
import { pool } from '@/lib/db';
import { log } from '@/lib/logger';

interface DecisionInput {
  id: string;
  decision: 'good' | 'bad';
}

async function handler(req: NextApiRequest, res: NextApiResponse): Promise<void> {
  if (req.method !== 'POST') {
    return apiResponse.methodNotAllowed(res, req.method ?? 'UNKNOWN', ['POST']);
  }

  const decisions = req.body?.decisions as DecisionInput[] | undefined;
  if (!Array.isArray(decisions) || decisions.length === 0) {
    return apiResponse.badRequest(res, 'decisions must be a non-empty array');
  }
  if (decisions.length > 100) {
    return apiResponse.badRequest(res, 'decisions cannot exceed 100 per request');
  }

  let updated = 0;
  for (const d of decisions) {
    if (!d.id || (d.decision !== 'good' && d.decision !== 'bad')) continue;
    try {
      const result = await pool.query(
        `UPDATE vlm_visual_photo_examples
         SET label = $1
         WHERE id = $2 AND job_type = 'civils'`,
        [d.decision === 'good' ? 'positive' : 'negative', d.id]
      );
      updated += result.rowCount ?? 0;
    } catch (err) {
      log.error('[CivilGallery] Error updating decision', { id: d.id, error: String(err) });
    }
  }

  log.info(`[CivilGallery] Updated ${updated} decisions`, undefined, 'CivilGallery');
  return apiResponse.success(res, { updated });
}

import { log } from '@/lib/logger';
import type { SiteCamStep } from '../lib/sitecamSteps';

const MODULE = 'useSiteCamCapture';

export type StepStatus = 'pending' | 'validating' | 'pass' | 'fail' | 'escalated';

export interface StepState {
  stepNumber: number;
  label: string;
  hasVlm: boolean;
  status: StepStatus;
  photoBase64: string | null;
  attemptNumber: number;
  failReasons: string[];
  corrections: string[];
}

export interface SiteInfo {
  jobType: 'activations' | 'civils';
  siteId: string;
  customerName: string | null;
  address: string | null;
  projectName: string | null;
}

function fileToBase64(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => {
      const result = reader.result as string;
      // Strip data URL prefix — API expects raw base64
      resolve(result.split(',')[1] ?? result);
    };
    reader.onerror = () => reject(new Error('FileReader failed'));
    reader.readAsDataURL(file);
  });
}

export function useSiteCamCapture(steps: readonly SiteCamStep[], siteInfo: SiteInfo) {
  const [stepStates, setStepStates] = useState<StepState[]>(
    steps.map((s) => ({
      stepNumber: s.number,
      label: s.label,
      hasVlm: s.hasVlm,
      status: 'pending',
      photoBase64: null,
      attemptNumber: 0,
      failReasons: [],
      corrections: [],
    }))
  );
  const [currentStepIndex, setCurrentStepIndex] = useState(0);
  const [uploading, setUploading] = useState(false);
  const [uploadError, setUploadError] = useState<string | null>(null);
  const [uploadResult, setUploadResult] = useState<{ uploadedCount: number } | null>(null);

  const currentStep = stepStates[currentStepIndex] ?? null;
  const allDone = stepStates.every((s) => s.status === 'pass' || s.status === 'escalated');

  const captureAndValidate = useCallback(async (file: File): Promise<void> => {
    const idx = currentStepIndex;
    const step = stepStates[idx];
    if (!step) return;

    const base64 = await fileToBase64(file);
    const nextAttempt = step.attemptNumber + 1;

    setStepStates((prev) =>
      prev.map((s, i) =>
        i === idx
          ? { ...s, photoBase64: base64, status: 'validating', attemptNumber: nextAttempt, failReasons: [], corrections: [] }
          : s
      )
    );

    if (!step.hasVlm) {
      // No VLM check — auto-pass
      setStepStates((prev) =>
        prev.map((s, i) => (i === idx ? { ...s, status: 'pass' } : s))
      );
      setTimeout(() => setCurrentStepIndex((i) => Math.min(i + 1, steps.length - 1)), 1500);
      return;
    }

    try {
      const resp = await fetch('/api/sitecam/validate', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          jobType: siteInfo.jobType,
          stepNumber: step.stepNumber,
          siteId: siteInfo.siteId,
          photoBase64: base64,
          attemptNumber: nextAttempt,
        }),
      });

      if (!resp.ok) {
        log.warn('Validate API error', { status: resp.status }, MODULE);
        // On API error, allow pass to avoid blocking technician
        setStepStates((prev) =>
          prev.map((s, i) => (i === idx ? { ...s, status: 'pass' } : s))
        );
        setTimeout(() => setCurrentStepIndex((i) => Math.min(i + 1, steps.length - 1)), 1500);
        return;
      }

      const data = await resp.json() as {
        data?: { pass: boolean; reasons: string[]; corrections: string[]; maxAttempts: number };
      };
      const result = data.data;
      if (!result) return;

      if (result.pass) {
        setStepStates((prev) =>
          prev.map((s, i) => (i === idx ? { ...s, status: 'pass', photoBase64: base64 } : s))
        );
        setTimeout(() => setCurrentStepIndex((i) => Math.min(i + 1, steps.length - 1)), 1500);
      } else if (nextAttempt >= result.maxAttempts) {
        // 3rd fail — escalate
        await escalateStep(idx, base64, result.reasons);
      } else {
        setStepStates((prev) =>
          prev.map((s, i) =>
            i === idx
              ? { ...s, status: 'fail', failReasons: result.reasons, corrections: result.corrections }
              : s
          )
        );
      }
    } catch (err) {
      log.error('Validate fetch failed', { error: String(err) }, MODULE);
      // Network error — allow pass
      setStepStates((prev) =>
        prev.map((s, i) => (i === idx ? { ...s, status: 'pass' } : s))
      );
      setTimeout(() => setCurrentStepIndex((i) => Math.min(i + 1, steps.length - 1)), 1500);
    }
  }, [currentStepIndex, stepStates, siteInfo, steps.length]);

  const escalateStep = useCallback(async (idx: number, photoBase64: string, failReasons: string[]): Promise<void> => {
    const step = stepStates[idx];
    if (!step) return;

    try {
      await fetch('/api/sitecam/escalate', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          jobType: siteInfo.jobType,
          siteId: siteInfo.siteId,
          stepNumber: step.stepNumber,
          failReasons,
          attemptPhotos: [],
        }),
      });
    } catch (err) {
      log.warn('Escalate failed — marking locally anyway', { error: String(err) }, MODULE);
    }

    setStepStates((prev) =>
      prev.map((s, i) => (i === idx ? { ...s, status: 'escalated' } : s))
    );
    setTimeout(() => setCurrentStepIndex((i) => Math.min(i + 1, steps.length - 1)), 2000);
  }, [stepStates, siteInfo, steps.length]);

  const submitAll = useCallback(async (): Promise<void> => {
    setUploading(true);
    setUploadError(null);
    try {
      const photos = stepStates
        .filter((s) => s.photoBase64 && (s.status === 'pass' || s.status === 'escalated'))
        .map((s) => ({
          stepNumber: s.stepNumber,
          stepLabel: s.label,
          filename: `step-${s.stepNumber}.jpg`,
          base64: s.photoBase64!,
        }));

      const resp = await fetch('/api/sitecam/upload', {
        method: 'POST',
        credentials: 'include',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ jobType: siteInfo.jobType, siteId: siteInfo.siteId, photos }),
      });

      if (!resp.ok) throw new Error(`Upload failed: HTTP ${resp.status}`);
      const data = await resp.json() as { data?: { uploadedCount: number } };
      setUploadResult({ uploadedCount: data.data?.uploadedCount ?? photos.length });
    } catch (err) {
      setUploadError(err instanceof Error ? err.message : 'Upload failed');
    } finally {
      setUploading(false);
    }
  }, [stepStates, siteInfo]);

  return {
    stepStates,
    currentStep,
    currentStepIndex,
    allDone,
    uploading,
    uploadError,
    uploadResult,
    captureAndValidate,
    submitAll,
  };
}

      <MyPortalShell title="SiteCam" showFooterNav={false}>
        <div className="flex items-center justify-center pt-24 gap-2 text-sm text-neutral-400">
          <Loader2 className="w-4 h-4 animate-spin" />
          Loading…
        </div>
      </MyPortalShell>
    );
  }

  if (state === 'guest') {
    if (typeof window !== 'undefined') void router.replace('/my');
    return <MyPortalShell title="SiteCam" showFooterNav={false}><div /></MyPortalShell>;
  }

  if (siteError || !siteInfo || !profile) {
    return (
      <MyPortalShell title="SiteCam" showFooterNav={false}>
        <div className="flex items-start gap-2 rounded-lg bg-red-950/50 border border-red-800 px-3 py-3 text-sm text-red-200 mt-4">
          <AlertCircle className="w-4 h-4 mt-0.5 shrink-0" />
          {siteError ?? 'Could not load site.'}
        </div>
      </MyPortalShell>
    );
  }

  return <SiteCamWizard profile={profile} siteInfo={siteInfo} />;
};
